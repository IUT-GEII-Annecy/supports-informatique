\subsection{Une première attaque}

\begin{UPSTIManipulation}{Étape 4 — Subir des dégâts}
  \textbf{Contexte :} On veut faire subir des dégâts à un personnage, sans jamais faire
  descendre ses PV en dessous de 0.

  \textbf{Cahier des charges :}
  \begin{itemize}
    \item Écrire une fonction \texttt{Personnage subir\_degats(Personnage cible, int degats)}
      qui renvoie \texttt{cible} avec ses PV réduits de \texttt{degats}, sans descendre en
      dessous de 0
    \item Dans le \texttt{main}, afficher la fiche du personnage, lui faire subir des dégâts
      (valeur demandée à l'utilisateur), puis afficher sa fiche à nouveau
  \end{itemize}

  \textbf{Exemple d'exécution :}
  \begin{lstlisting}[language=bash]
Nom du personnage: Arthor
Classe (0=Guerrier, 1=Mage, 2=Archer): 0
Avant : Arthor (PV: 35, ATQ: 7, DEF: 6)
Degats subis: 40
Apres : Arthor (PV: 0, ATQ: 7, DEF: 6)
  \end{lstlisting}

  \UPSTIetape{Modifier \texttt{jeu\_combat.c} pour respecter le cahier des charges ci-dessus.}
\end{UPSTIManipulation}

\begin{UPSTIManipulation}{Étape 5 — Un combat en un contre un}
  \textbf{Contexte :} On fait s'affronter deux personnages, tour par tour, jusqu'à ce que l'un
  des deux soit vaincu (0 PV).

  \textbf{Cahier des charges :}
  \begin{itemize}
    \item Créer deux personnages \texttt{p1} et \texttt{p2}
    \item Tant qu'aucun des deux n'a 0 PV : \texttt{p1} attaque \texttt{p2}, puis si
      \texttt{p2} est encore en vie, \texttt{p2} attaque \texttt{p1}
    \item Afficher les PV restants des deux personnages après chaque tour
    \item À la fin du combat, afficher le nom du vainqueur
  \end{itemize}

  \textbf{Exemple d'exécution :}
  \begin{lstlisting}[language=bash]
Tour 1 : Arthor PV=25 / Lysia PV=29
Tour 2 : Arthor PV=25 / Lysia PV=16
Tour 3 : Arthor PV=18 / Lysia PV=16
Tour 4 : Arthor PV=18 / Lysia PV=3
Tour 5 : Arthor PV=11 / Lysia PV=0
Vainqueur : Arthor
  \end{lstlisting}

  \UPSTIetape{Modifier \texttt{jeu\_combat.c} pour respecter le cahier des charges ci-dessus.}
\end{UPSTIManipulation}

\subsection{Une équipe de héros}

\begin{UPSTIManipulation}{Étape 6 — Constituer une équipe}
  \textbf{Contexte :} Un seul héros, c'est bien, mais une équipe, c'est mieux. On remplace le
  combat 1 contre 1 par la constitution d'une équipe entière.

  \textbf{Cahier des charges :}
  \begin{itemize}
    \item Constituer un tableau de 3 personnages (\texttt{Personnage equipe[3]}), remplis l'un
      après l'autre (nom, classe)
    \item Écrire une fonction qui affiche la fiche de chaque personnage de l'équipe
    \item Écrire une fonction qui renvoie la somme des PV de toute l'équipe
    \item Afficher l'équipe, puis ses PV totaux
  \end{itemize}

  \textbf{Exemple d'exécution :}
  \begin{lstlisting}[language=bash]
Personnage 0 - Nom: Arthor
Personnage 0 - Classe (0=Guerrier, 1=Mage, 2=Archer): 0
Personnage 1 - Nom: Lysia
Personnage 1 - Classe (0=Guerrier, 1=Mage, 2=Archer): 1
Personnage 2 - Nom: Finn
Personnage 2 - Classe (0=Guerrier, 1=Mage, 2=Archer): 2

Arthor (PV: 35, ATQ: 7, DEF: 6)
Lysia (PV: 20, ATQ: 10, DEF: 2)
Finn (PV: 25, ATQ: 9, DEF: 3)
PV totaux de l'equipe: 80
  \end{lstlisting}

  \textbf{Remarque :} comme pour un tableau "classique", la taille de l'équipe doit toujours
  être transmise en paramètre aux fonctions qui le parcourent.

  \UPSTIetape{Modifier \texttt{jeu\_combat.c} pour respecter le cahier des charges ci-dessus.}
\end{UPSTIManipulation}

\begin{UPSTIManipulation}{Étape 7 — Face à plusieurs monstres}
  \textbf{Contexte :} L'équipe de héros affronte maintenant un \textbf{groupe} de monstres (et
  non un seul), représenté lui aussi par un tableau de structures \texttt{Personnage}.

  \textbf{Cahier des charges :}
  \begin{itemize}
    \item Constituer un second tableau de 2 personnages représentant les monstres, rempli de la
      même façon que l'équipe de héros
    \item Afficher le groupe de monstres
    \item Faire subir à \texttt{equipe[0]} les attaques successives de chaque monstre encore en
      vie, et afficher son état après chaque attaque
  \end{itemize}

  \textbf{Exemple d'exécution :}
  \begin{lstlisting}[language=bash]
--- Equipe de heros ---
Arthor (PV: 35, ATQ: 7, DEF: 6)
Lysia (PV: 20, ATQ: 10, DEF: 2)
Finn (PV: 25, ATQ: 9, DEF: 3)

--- Monstres ---
Gobelin (PV: 15, ATQ: 5, DEF: 1)
Orc (PV: 30, ATQ: 8, DEF: 4)

Arthor subit l'attaque de Gobelin : PV restants 30
Arthor subit l'attaque de Orc : PV restants 22
  \end{lstlisting}

  \UPSTIetape{Modifier \texttt{jeu\_combat.c} pour respecter le cahier des charges ci-dessus.}
\end{UPSTIManipulation}
