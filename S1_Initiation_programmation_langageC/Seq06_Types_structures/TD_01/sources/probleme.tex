\begin{UPSTIexercice}{Projet d'ensemble : bulletins d'une classe}
    On souhaite écrire un petit programme qui gère les \textbf{bulletins} d'une classe.
    Chaque élève a plusieurs notes dans différentes matières, avec un \textbf{coefficient}
    pour chaque matière.

    On suppose que :
    \begin{itemize}
        \item tous les élèves ont les \textbf{mêmes matières}, dans le même ordre,
        \item il y a au maximum \texttt{30} élèves dans la classe,
        \item il y a par exemple \texttt{4} matières par élève,
        \item chaque élève peut avoir \textbf{plusieurs notes} par matière (par ex. contrôles, devoirs, etc.).
    \end{itemize}

    \UPSTIquestion{Compléter les constantes et les types structurés permettant de modéliser ce problème : une structure \texttt{T\_Matiere} (nom, coefficient) et une structure \texttt{T\_Eleve} (prénom, âge, notes par matière, et le nombre de notes réellement saisies par matière).}

    \begin{lstlisting}[language=c,escapeinside={(*}{*)},caption={Modélisation, à compléter}]
#define TAILLE_MAX_CLASSE 30
#define NB_MATIERES       4
#define NB_NOTES_MAX      10

typedef struct {
    (*\trou[string nom;]{7.5cm}*)
    (*\trou[float coeff;]{7.5cm}*)
} T_Matiere;

typedef struct {
    (*\trou[{float notes[NB\_MATIERES][NB\_NOTES\_MAX];}]{11.5cm}*)
    (*\trou[{int nb\_notes[NB\_MATIERES];}]{11.5cm}*)  // nombre de notes pour chaque matiere
    (*\trou[string prenom;]{7.5cm}*)
    (*\trou[int age;]{7.5cm}*)
} T_Eleve;

T_Matiere matieres[NB_MATIERES];
\end{lstlisting}

    Cette modélisation sépare les définitions des matières (nom et coefficient, partagés par tous)
    du stockage des notes individuelles de chaque élève. Chaque élève peut avoir un nombre
    différent de notes par matière, stocké dans \texttt{nb\_notes[]}.

    % ----------------------------------------------------------------------

    \UPSTIquestion{Écrire le corps de la fonction \texttt{float moyenne\_eleve(T\_Eleve e, T\_Matiere matieres[])} qui renvoie la moyenne générale de l'élève, en tenant compte des coefficients de chaque matière.}

    \begin{lstlisting}[language=c,caption={Prototype}]
float moyenne_eleve(T_Eleve e, T_Matiere matieres[]);
\end{lstlisting}

    \UPSTIquadrillage{7}

\begin{UPSTIprofOnlyEnv}
\begin{UPSTIcorrectionP}{moyenne\_eleve}
\begin{lstlisting}[language=c,caption={Correction}]
float moyenne_eleve(T_Eleve e, T_Matiere matieres[]) {
    int i, k;
    float somme_notes = 0.0f;
    float somme_coeffs = 0.0f;

    for (i = 0; i < NB_MATIERES; i++) {
        float moy_matiere = 0.0f;

        // Calculer la moyenne pour cette matiere
        for (k = 0; k < e.nb_notes[i]; k++) {
            moy_matiere += e.notes[i][k];
        }
        if (e.nb_notes[i] > 0) {
            moy_matiere /= e.nb_notes[i];
        }

        // Ajouter a la moyenne ponderee
        somme_notes  += moy_matiere * matieres[i].coeff;
        somme_coeffs += matieres[i].coeff;
    }

    if (somme_coeffs == 0.0f) {
        return 0.0f;
    }
    return somme_notes / somme_coeffs;
}
\end{lstlisting}
\end{UPSTIcorrectionP}
\end{UPSTIprofOnlyEnv}

    % ----------------------------------------------------------------------

    \UPSTIquestion{Écrire le corps de la fonction \texttt{float moyenne\_matiere(T\_Eleve classe[], int nbEleves, int indiceMatiere)} qui renvoie la moyenne de la classe pour une matière donnée (identifiée par son indice). Pourquoi cette fonction prend-elle \texttt{nbEleves} en paramètre ?}

    Comme pour un tableau de structures classique, le tableau \texttt{classe} ne connaît pas lui-même
    le nombre d'éléments réellement utilisés \trou[(TAILLE\_MAX\_CLASSE est une taille maximale, pas le nombre d'élèves effectivement saisis)]{12.5cm}.

    \begin{lstlisting}[language=c,caption={Prototype}]
float moyenne_matiere(T_Eleve classe[], int nbEleves, int indiceMatiere);
\end{lstlisting}

    \UPSTIquadrillage{6}

\begin{UPSTIprofOnlyEnv}
\begin{UPSTIcorrectionP}{moyenne\_matiere}
\begin{lstlisting}[language=c,caption={Correction}]
float moyenne_matiere(T_Eleve classe[], int nbEleves, int indiceMatiere) {
    int i, k;
    float somme = 0.0f;
    int nb_notes_total = 0;

    for (i = 0; i < nbEleves; i++) {
        for (k = 0; k < classe[i].nb_notes[indiceMatiere]; k++) {
            somme += classe[i].notes[indiceMatiere][k];
            nb_notes_total++;
        }
    }

    if (nb_notes_total == 0) {
        return 0.0f;
    }
    return somme / nb_notes_total;
}
\end{lstlisting}
\end{UPSTIcorrectionP}
\end{UPSTIprofOnlyEnv}

    % ----------------------------------------------------------------------

    \UPSTIquestion{On va maintenant écrire une partie de \texttt{main}. On dispose d'un tableau
        \texttt{T\_Eleve classe[TAILLE\_MAX\_CLASSE];} et d'un tableau \texttt{T\_Matiere matieres[NB\_MATIERES];}.
        Compléter le code permettant de saisir le nombre d'élèves, les matières, puis pour chaque élève son prénom, son âge et ses notes, et enfin d'afficher la moyenne de chaque élève.}

    \begin{lstlisting}[language=c,escapeinside={(*}{*)},caption={main — saisie et affichage, à compléter}]
int main(void) {
    T_Eleve classe[TAILLE_MAX_CLASSE];
    T_Matiere matieres[NB_MATIERES];
    int nbEleves;
    int i, j, k;

    // 1. Saisie du nombre d'eleves
    do {
        nbEleves = get_int("Nombre d'eleves (1..%d) : ", TAILLE_MAX_CLASSE);
    } while (nbEleves < 1 || nbEleves > TAILLE_MAX_CLASSE);

    // 2. Saisie des matieres
    for (j = 0; j < NB_MATIERES; j++) {
        printf("Matiere %d\n", j + 1);
        (*\trou[{matieres[j].nom}]{7.5cm}*)   = get_string("Nom de la matiere : ");
        (*\trou[{matieres[j].coeff}]{7.5cm}*) = get_float("Coefficient : ");
    }
    printf("\n");

    // 3. Saisie des eleves et de leurs notes
    for (i = 0; i < nbEleves; i++) {
        printf("Eleve %d\n", i + 1);
        (*\trou[{classe[i].prenom}]{7.5cm}*) = get_string("Prenom : ");
        (*\trou[{classe[i].age}]{7.5cm}*)    = get_int("Age : ");

        for (j = 0; j < NB_MATIERES; j++) {
            (*\trou[{classe[i].nb\_notes[j]}]{9.5cm}*) = get_int(
                "Nombre de notes en %s : ", matieres[j].nom);

            for (k = 0; k < classe[i].nb_notes[j]; k++) {
                (*\trou[{classe[i].notes[j][k]}]{9.5cm}*) = get_float(
                    "Note %d : ", k + 1);
            }
        }
        printf("\n");
    }

    // 4. Affichage des moyennes
    for (i = 0; i < nbEleves; i++) {
        float m = (*\trou[{moyenne\_eleve(classe[i], matieres)}]{11.5cm}*);
        printf("Moyenne de %s = %.2f\n", classe[i].prenom, m);
    }

    return 0;
}
\end{lstlisting}

    % ----------------------------------------------------------------------

    \UPSTIquestion{À partir des fonctions précédentes, compléter le code (dans le \texttt{main} ou dans une fonction dédiée) qui détermine l'élève ayant la meilleure moyenne générale, calcule la moyenne de la classe pour chaque matière, et affiche la matière dont la moyenne de classe est la plus faible (matière « à renforcer »).}

    \begin{lstlisting}[language=c,escapeinside={(*}{*)},caption={Meilleur élève et matière la plus faible, à compléter}]
int indice_meilleur = 0;
float meilleure_moy = moyenne_eleve(classe[0], matieres);
float m;
int i;

for (i = 1; i < nbEleves; i++) {
    m = moyenne_eleve(classe[i], matieres);
    if ((*\trou[{m > meilleure\_moy}]{9.5cm}*)) {
        meilleure_moy = m;
        indice_meilleur = i;
    }
}

printf("Meilleur eleve : %s (moyenne = %.2f)\n",
       (*\trou[{classe[indice\_meilleur].prenom}]{7.5cm}*), meilleure_moy);

float moyennesMatieres[NB_MATIERES];
int j;

for (j = 0; j < NB_MATIERES; j++) {
    moyennesMatieres[j] = (*\trou[{moyenne\_matiere(classe, nbEleves, j)}]{13.5cm}*);
    printf("Moyenne de la classe en %s : %.2f\n",
           matieres[j].nom, moyennesMatieres[j]);
}

// Recherche de la matiere avec la plus faible moyenne
int indice_pire_matiere = 0;
float pire_moyenne = moyennesMatieres[0];

for (j = 1; j < NB_MATIERES; j++) {
    if ((*\trou[{moyennesMatieres[j] < pire\_moyenne}]{11.5cm}*)) {
        pire_moyenne = moyennesMatieres[j];
        indice_pire_matiere = j;
    }
}

printf("Matiere a renforcer : %s (moyenne = %.2f)\n",
       (*\trou[{matieres[indice\_pire\_matiere].nom}]{9.5cm}*), pire_moyenne);
\end{lstlisting}

    Cet ensemble de questions mobilise plusieurs \textbf{structures} (données partagées et individuelles),
    des \textbf{tableaux de structures}, des \textbf{fonctions} qui prennent des structures et des tableaux
    de structures en paramètre, ainsi que des \textbf{boucles imbriquées} et un peu d'algorithmique
    (recherche de max/min).
\end{UPSTIexercice}
