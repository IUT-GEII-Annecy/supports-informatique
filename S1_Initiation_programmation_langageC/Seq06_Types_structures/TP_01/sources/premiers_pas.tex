\subsection{Une première structure}

\begin{UPSTIManipulation}{Étape 1 — Fiche de personnage simple}
  \textbf{Contexte :} On veut représenter un personnage de jeu par son nom, ses points de vie
  et sa puissance d'attaque.

  \textbf{Cahier des charges :}
  \begin{itemize}
    \item Déclarer une structure \texttt{Personnage} avec les champs : \texttt{string nom},
      \texttt{int pv} et \texttt{int attaque}
    \item Dans le \texttt{main}, créer une variable \texttt{héros} de type \texttt{Personnage}
    \item Demander à l'utilisateur son nom, ses pv et son attaque, avec les fonctions
      \texttt{get\_string} et \texttt{get\_int} de la librairie \texttt{cs50} (ajouter
      \texttt{\#include <cs50.h>} en haut du fichier)
    \item Afficher sa fiche sous la forme : \texttt{Nom (PV: x, ATQ: y)}
  \end{itemize}

  \textbf{Exemple d'exécution :}
  \begin{lstlisting}[language=bash]
Nom du personnage: Arthor
Points de vie: 30
Attaque: 8
-> Arthor (PV: 30, ATQ: 8)
  \end{lstlisting}

  \textbf{Squelette à copier-coller :} collez le bloc suivant dans la section
  \texttt{// === Déclaration des structures ===} du fichier, puis complétez les \texttt{TODO}.
  Complétez également le \texttt{main} (dans \texttt{// === Programme principal ===}) pour
  déclarer \texttt{héros}, demander les trois informations avec \texttt{get\_string} /
  \texttt{get\_int}, et afficher sa fiche.

  \begin{lstlisting}[language=C]
typedef struct {
    string nom;   // TODO : completer avec pv et attaque
} Personnage;
  \end{lstlisting}

  \UPSTIetape{Compléter le squelette \texttt{jeu\_combat.c} pour respecter le cahier des charges ci-dessus.}
\end{UPSTIManipulation}

\subsection{Structure imbriquée}

\begin{UPSTIManipulation}{Étape 2 — Séparer les statistiques}
  \textbf{Contexte :} On souhaite ajouter une statistique de \textbf{défense}, et regrouper
  les statistiques combattantes dans une structure \texttt{Stats} dédiée, \textbf{imbriquée}
  dans \texttt{Personnage}.

  \textbf{Cahier des charges :}
  \begin{itemize}
    \item Déclarer une structure \texttt{Stats} avec les champs \texttt{int pv},
      \texttt{int attaque}, \texttt{int defense}
    \item \texttt{Personnage} n'a plus directement les champs \texttt{pv} et \texttt{attaque} :
      il contient désormais un champ \texttt{Stats stats}
    \item Écrire une fonction \texttt{void afficher\_personnage(Personnage p)} qui affiche la
      fiche sous la forme \texttt{Nom (PV: x, ATQ: y, DEF: z)}
    \item Adapter la saisie pour demander la défense en plus, puis afficher la fiche avec cette
      nouvelle fonction
  \end{itemize}

  \textbf{Exemple d'exécution :}
  \begin{lstlisting}[language=bash]
Nom du personnage: Lysia
Points de vie: 25
Attaque: 6
Defense: 4
-> Lysia (PV: 25, ATQ: 6, DEF: 4)
  \end{lstlisting}

  \textbf{Squelettes à copier-coller :} la structure \texttt{Stats} va dans la section
  \texttt{// === Déclaration des structures ===} (\textbf{avant} \texttt{Personnage}, qui
  l'utilise). Le prototype de \texttt{afficher\_personnage} va dans
  \texttt{// === Déclaration des fonctions ===}, et son corps dans
  \texttt{// === Définition des fonctions ===}.

  \begin{lstlisting}[language=C]
typedef struct {
    int pv;
    int attaque;
    int defense;   // TODO : relire la structure Personnage pour l'adapter
} Stats;

{

\subsection{Type énuméré}

\begin{UPSTIinfor}{Rappel}
  Avant de poursuivre, relisez dans le cours la partie sur les \textbf{types énumérés}
  (\texttt{enum}) : leur déclaration, et leur utilisation dans un \texttt{switch}.
\end{UPSTIinfor}

\begin{UPSTIManipulation}{Étape 3 — Choisir une classe}
  \textbf{Contexte :} Chaque personnage appartient désormais à une \textbf{classe}
  (Guerrier, Mage ou Archer), qui détermine ses statistiques de départ.

  \textbf{Cahier des charges :}
  \begin{itemize}
    \item Déclarer un type énuméré \texttt{Classe} : \texttt{GUERRIER}, \texttt{MAGE},
      \texttt{ARCHER}
    \item Ajouter un champ \texttt{Classe classe} à \texttt{Personnage}
    \item Écrire une fonction \texttt{Stats stats\_initiales(Classe c)} qui renvoie les
      statistiques de départ correspondant à la classe :
      \begin{itemize}
        \item \texttt{GUERRIER} : PV=35, ATQ=7, DEF=6
        \item \texttt{MAGE} : PV=20, ATQ=10, DEF=2
        \item \texttt{ARCHER} : PV=25, ATQ=9, DEF=3
      \end{itemize}
    \item Remplacer la saisie manuelle des statistiques par la demande du \textbf{nom} et de la
      \textbf{classe} (entier 0, 1 ou 2), les statistiques étant alors initialisées
      automatiquement
  \end{itemize}

  \textbf{Exemple d'exécution :}
  \begin{lstlisting}[language=bash]
Nom du personnage: Arthor
Classe (0=Guerrier, 1=Mage, 2=Archer): 0
-> Arthor (PV: 35, ATQ: 7, DEF: 6)
  \end{lstlisting}

  \textbf{Squelettes à copier-coller :} l'énuméré \texttt{Classe} va dans
  \texttt{// === Déclaration des structures ===}, \textbf{avant} \texttt{Personnage} qui
  l'utilise. Le prototype de \texttt{stats\_initiales} va avec les autres prototypes, son corps
  avec les autres corps de fonctions.

  \begin{lstlisting}[language=C]
typedef enum {
    GUERRIER,
    MAGE,
    ARCHER
} Classe;
  \end{lstlisting}

// Prototype
Stats stats_initiales(Classe c);

{
    Stats s;

        // TODO : un case par valeur de Classe, avec les PV/ATQ/DEF de depart
    }

  \end{lstlisting}

  \UPSTIetape{Modifier \texttt{jeu\_combat.c} pour respecter le cahier des charges ci-dessus.}
\end{UPSTIManipulation}
