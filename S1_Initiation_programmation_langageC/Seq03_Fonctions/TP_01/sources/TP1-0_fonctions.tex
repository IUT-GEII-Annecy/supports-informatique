\begin{UPSTIexercice}{Une fonction qui demande un entier positif}
	\UPSTIquestion Ecrire une fonction \texttt{int demander\_entier\_positif(void)} qui demande à l'utilisateur de saisir un entier positif, et qui répète la demande tant que l'utilisateur n'entre pas un entier positif.
	\UPSTIquestion Utiliser cette fonction dans un programme pour récupérer un entier positif et l'afficher.
\end{UPSTIexercice}

\begin{UPSTIexercice}{Une fonction qui affiche une ligne}
	Reprendre le programme carre qui affiche un carré d'étoiles.\\
	\UPSTIquestion Créer une fonction \texttt{void afficher\_ligne(int n)} qui affiche une ligne de \texttt{n} étoiles.
	\UPSTIquestion Utiliser cette fonction dans le programme principal pour afficher un carré dont les dimensions seront demandées à l'utilisateur.
\end{UPSTIexercice}

\begin{UPSTIexercice}{Afficher des espaces}
	\UPSTIquestion Ecrire une fonction \texttt{void afficher\_espaces(int n)} qui affiche \texttt{n} espaces.
	\UPSTIquestion Utiliser cette fonction et la précédente pour afficher un triangle d'étoiles isocèle et centré, de hauteur~\texttt{h} (entier entré par l'utilisateur).
	\textit{Vous pouvez vous aider de l'exercice \ref{boucles:ex:triangle_etoile} du TD sur les boucles.}
\end{UPSTIexercice}

\begin{UPSTIexercice}{Débuggeur}

\end{UPSTIexercice}

\begin{UPSTIexercice}{Calculer le maximum}
	Ecrire une fonction \texttt{maximum} qui prend en entrée deux paramètres : \texttt{a} et \texttt{b} et qui retourne la valeur maximale entre ces deux nombres.

	Méthode : 
	\begin{enumerate}
		\item Dessiner la boite représentant la fonction
		\item Ecrire le prototype de la fonction
		\item Ecrire le corps de la fonction.
	\end{enumerate}
\end{UPSTIexercice}

\begin{UPSTIexercice}{Températures}
	Ecrire deux fonctions \texttt{celcius\_vers\_fahrenheit} et \texttt{fahrenheit\_vers\_celcius} qui réalisent les conversions de températures.
\end{UPSTIexercice}
