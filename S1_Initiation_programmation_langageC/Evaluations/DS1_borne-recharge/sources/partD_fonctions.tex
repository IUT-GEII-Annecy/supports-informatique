\element{fonc-ecrire-mode}{
    \begin{question}{fonc-ecrire-mode}
        On souhaite la fonction
        \lstinline[language=C]|int determiner_mode(float puissance_kw, int cable_verrouille)|
        qui retourne un \textbf{code entier} caractérisant la décision de la borne :
        \begin{itemize}\setlength{\itemsep}{0pt}
            \item -1 si le câble n'est pas verrouillé ;
            \item 0 si la puissance demandée ne dépasse pas 22 kW (mode lent) ;
            \item 1 si la puissance demandée est comprise entre 22 kW (exclu) et 50 kW (inclus) (mode rapide) ;
            \item 2 si la puissance demandée dépasse 50 kW sans dépasser 150 kW (mode ultra-rapide) ;
            \item 3 si la puissance demandée dépasse 150 kW (demande hors domaine de la borne).
        \end{itemize}

        La fonction ne doit rien afficher et doit retourner une valeur dans \textbf{tous} les cas.

        Ecrire la définition (le corps) de cette fonction.
        \evaluationProfGrille{3}{16}

        
    \end{question}
}
