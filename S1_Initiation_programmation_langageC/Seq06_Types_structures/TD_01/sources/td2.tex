\begin{UPSTIexercice}{Le problème sans structure}
    On souhaite gérer les informations d’un étudiant :
    \begin{itemize}
        \item son prénom,
        \item son âge,
        \item sa moyenne.
    \end{itemize}

    \UPSTIquestion{Comment stocker les informations de 3 étudiants différents, sans utiliser de structure ?}

    Sans structure, on peut multiplier les variables (\texttt{prenom1}, \texttt{age1}, \texttt{moyenne1}, \texttt{prenom2}, ...)
    ou utiliser \trou[{3 tableaux parallèles (prenoms[], ages[], moyennes[])}]{10cm}.

    \UPSTIquestion{Pourquoi cette approche devient-elle source d'erreurs ?}

    Parce qu'il faut constamment \trou[se souvenir que les éléments correspondant au même étudiant (par ex. prenom1, age1, moyenne1, ou les cases d'indice i dans chaque tableau) vont ensemble]{13cm} :
    on risque de mélanger les indices ou d'oublier un champ, car on ne manipule jamais « un étudiant » mais plusieurs variables séparées.
\end{UPSTIexercice}

\begin{UPSTIinfor}{Les types structurés en C}
    En C, on peut regrouper plusieurs variables dans une seule entité appelée \textbf{structure}, définie avec le mot-clé \texttt{struct}.

    Dans notre cas, nous allons créer un type structuré qui contiendra différentes informations concernant une même entité.

    La définition d'un type structuré se fait généralement avec le mot-clé \texttt{typedef} pour simplifier son utilisation ultérieure. La syntaxe générale est la suivante :
    \begin{lstlisting}[language=C]
typedef struct {
    type1 champ1;
    type2 champ2;
    ...
    typeN champN;
} NomDeLaStructure;
\end{lstlisting}

    Prenons l'exemple d'un contact dans un carnet d'adresses. On peut définir une structure \texttt{Contact} qui regroupe plusieurs champs :
    \begin{lstlisting}[language=C]
typedef struct {
    string nom;
    string prenom;
    string telephone;
    string email;
    int anneeNaissance;
} T_Contact;
\end{lstlisting}
\end{UPSTIinfor}

\begin{UPSTIexercice}{Créer une structure cohérente}
    On regroupe les données dans une seule entité : un \textbf{type structuré}.

    \UPSTIquestion{Compléter la définition du type \texttt{T\_Etudiant} (prénom, âge, moyenne), les déclarations d'une variable \texttt{etudiant\_1} et d'un tableau \texttt{classe} de 30 étudiants, puis la boucle de saisie des 30 étudiants et la recherche du meilleur étudiant de la classe.}

    \begin{lstlisting}[language=c,escapeinside={(*}{*)},caption={T\_Etudiant, à compléter}]
typedef struct {
    (*\trou[string prenom;]{6cm}*)
    (*\trou[int age;]{6cm}*)
    (*\trou[float moyenne;]{6cm}*)
} T_Etudiant;

T_Etudiant etudiant_1;
(*\trou[{T\_Etudiant classe[30];}]{8cm}*)
\end{lstlisting}

    \begin{lstlisting}[language=c,escapeinside={(*}{*)},caption={Saisie et recherche du meilleur étudiant, à compléter}]
int i;
for (i = 0; i < 30; i++) {
    printf("Etudiant %d\n", i + 1);
    (*\trou[{classe[i].prenom}]{6cm}*) = get_string("Prenom : ");
    (*\trou[{classe[i].age}]{6cm}*) = get_int("Age : ");
    (*\trou[{classe[i].moyenne}]{6cm}*) = get_float("Moyenne : ");
}

int indice_meilleur = 0;
for (i = 1; i < 30; i++) {
    if ((*\trou[{classe[i].moyenne > classe[indice\_meilleur].moyenne}]{10cm}*)) {
        indice_meilleur = i;
    }
}

printf("Meilleur etudiant : %s, moyenne = %.2f\n",
       (*\trou[{classe[indice\_meilleur].prenom}]{6cm}*),
       (*\trou[{classe[indice\_meilleur].moyenne}]{6cm}*));
\end{lstlisting}

    On peut initialiser un étudiant de deux façons : soit par initialisation directe
    \trou[{T\_Etudiant etudiant\_1 = \{"Bob", 22, 15.5f\};}]{10cm}, soit champ par champ
    (\texttt{etudiant\_1.prenom = "Bob";}, etc.).
\end{UPSTIexercice}

\begin{UPSTIexercice}{Représenter une heure}
    On veut manipuler une heure sous la forme \texttt{HH:MM:SS}.

    \UPSTIquestion{Compléter le type \texttt{T\_Heure} et les fonctions permettant d'afficher une heure, de la convertir en secondes, et inversement de reconstruire une heure à partir d'un nombre de secondes.}

    \begin{lstlisting}[language=c,escapeinside={(*}{*)},caption={T\_Heure, à compléter}]
typedef struct {
    (*\trou[int h;]{5cm}*)
    (*\trou[int m;]{5cm}*)
    (*\trou[int s;]{5cm}*)
} T_Heure;

void afficher_heure(T_Heure t) {
    printf("%02d:%02d:%02d\n", (*\trou[t.h, t.m, t.s]{6cm}*));
}

int en_secondes(T_Heure t) {
    return (*\trou[{t.h * 3600 + t.m * 60 + t.s}]{10cm}*);
}

T_Heure inverse(int total_secondes) {
    T_Heure t;
    (*\trou[{t.h = total\_secondes / 3600;}]{10cm}*)
    total_secondes %= 3600;
    (*\trou[{t.m = total\_secondes / 60;}]{10cm}*)
    (*\trou[{t.s = total\_secondes \% 60;}]{10cm}*)
    return t;
}
\end{lstlisting}

    \begin{lstlisting}[language=c,escapeinside={(*}{*)},caption={Utilisation dans le main, à compléter}]
int main(void) {
    T_Heure h1 = {8, 30, 0};
    T_Heure h2 = {10, 15, 30};

    int s1 = (*\trou[en\_secondes(h1)]{6cm}*);
    int s2 = (*\trou[en\_secondes(h2)]{6cm}*);

    printf("Difference = %d secondes\n", s2 - s1);
    return 0;
}
\end{lstlisting}
\end{UPSTIexercice}

\begin{UPSTIexercice}{Gérer un produit en stock}
    On veut garder les informations suivantes pour chaque produit :
    \begin{itemize}
        \item son nom (chaîne de caractères),
        \item son prix HT (nombre à virgule),
        \item son taux de TVA (nombre à virgule).
    \end{itemize}

    \UPSTIquestion{Compléter le type \texttt{T\_Produit} et la fonction renvoyant le prix TTC d'un produit (la fonction ne prendra qu'un seul argument).}

    \begin{lstlisting}[language=c,escapeinside={(*}{*)},caption={T\_Produit, à compléter}]
typedef struct {
    (*\trou[string nom;]{6cm}*)
    (*\trou[float prixHT;]{6cm}*)
    (*\trou[float tva;]{6cm}*)  // en pourcentage, par ex. 20.0
} T_Produit;

float prixTTC(T_Produit p) {
    return (*\trou[{p.prixHT * (1.0f + p.tva / 100.0f)}]{12cm}*);
}
\end{lstlisting}
\end{UPSTIexercice}

\begin{UPSTIexercice}{Gestion d’un vecteur 2D}
    On manipule des vecteurs dans un plan :
    \begin{lstlisting}[language=C]
typedef struct{
    float x;
    float y;
} T_Vecteur;
\end{lstlisting}

    \UPSTIquestion{Compléter les fonctions \texttt{norme}, \texttt{addition} et \texttt{oppose}, ainsi que leur utilisation dans le \texttt{main}.}

    \begin{lstlisting}[language=c,escapeinside={(*}{*)},caption={Fonctions sur T\_Vecteur, à compléter}]
#include <math.h>

float norme(T_Vecteur v) {
    return (*\trou[{sqrtf(v.x * v.x + v.y * v.y)}]{10cm}*);
}

T_Vecteur addition(T_Vecteur a, T_Vecteur b) {
    T_Vecteur res;
    (*\trou[{res.x = a.x + b.x;}]{9cm}*)
    (*\trou[{res.y = a.y + b.y;}]{9cm}*)
    return res;
}

T_Vecteur oppose(T_Vecteur v) {
    T_Vecteur res;
    (*\trou[{res.x = -v.x;}]{9cm}*)
    (*\trou[{res.y = -v.y;}]{9cm}*)
    return res;
}
\end{lstlisting}

    \begin{lstlisting}[language=c,escapeinside={(*}{*)},caption={Utilisation dans le main, à compléter}]
int main(void) {
    T_Vecteur a = {3.0f, 4.0f};
    T_Vecteur b = {1.0f, 2.0f};

    printf("||a|| = %f\n", (*\trou[norme(a)]{6cm}*));
    printf("||b|| = %f\n", (*\trou[norme(b)]{6cm}*));

    T_Vecteur diff = (*\trou[{addition(a, oppose(b))}]{10cm}*);
    printf("||a-b|| = %f\n", norme(diff));
    return 0;
}
\end{lstlisting}
\end{UPSTIexercice}

\begin{UPSTIexercice}{Extension : struct imbriquées}
    Il est également possible de créer des structures imbriquées, c'est-à-dire d'utiliser une structure comme champ d'une autre structure. Par exemple, on peut définir une structure \texttt{Adresse} et l'utiliser dans une structure \texttt{Etudiant} :
    \begin{lstlisting}[language=C]
typedef struct {
    string rue;
    int codePostal;
    string ville;
} T_Adresse;

typedef struct {
    string prenom;
    int age;
    float moyenne;
    T_Adresse adresse;
} T_EtudiantAdresse;
\end{lstlisting}

    \UPSTIquestion{Quel est l'avantage de créer une structure imbriquée plutôt qu'ajouter 3 champs à \texttt{Etudiant} ?}

    On regroupe tous les champs liés à l'adresse dans une seule structure \texttt{T\_Adresse}, ce qui permet
    \trou[de réutiliser T\_Adresse pour d'autres structures (Professeur, Entreprise, etc.) et de faciliter les modifications ultérieures]{13cm}.

    \UPSTIquestion{Compléter l'accès à la ville d'un étudiant, ainsi que la fonction \texttt{habite\_ville}.}

    \begin{lstlisting}[language=c,escapeinside={(*}{*)},caption={Accès et fonction sur T\_EtudiantAdresse, à compléter}]
// Si e est une variable de type T_EtudiantAdresse :
string ville_e = (*\trou[{e.adresse.ville}]{6cm}*);

int habite_ville(T_EtudiantAdresse e, const char* ville) {
    return (*\trou[{strcmp(e.adresse.ville, ville) == 0}]{10cm}*);
}
\end{lstlisting}
\end{UPSTIexercice}

\begin{UPSTIexercice}{Pour aller plus loin : tableau d'étudiants avec adresse}
    On dispose maintenant d'une \textbf{promotion entière} d'étudiants, chacun avec son adresse
    (type \texttt{T\_EtudiantAdresse} défini précédemment). On fixe une taille maximale de promotion :
    \begin{lstlisting}[language=C]
#define TAILLE_MAX_PROMO 40
\end{lstlisting}

    \UPSTIquestion{Compléter la déclaration du tableau de la promotion, la saisie de ses \texttt{nbEtudiants} étudiants (prénom, âge, moyenne, puis les 3 champs de l'adresse), et la fonction qui compte combien d'étudiants habitent une ville donnée.}

    \begin{lstlisting}[language=c,escapeinside={(*}{*)},caption={Déclaration et saisie, à compléter}]
(*\trou[{T\_EtudiantAdresse promo[TAILLE\_MAX\_PROMO];}]{12cm}*)
int nbEtudiants;
int i;

nbEtudiants = get_int("Nombre d'etudiants : ");

for (i = 0; i < nbEtudiants; i++) {
    promo[i].prenom = get_string("Prenom : ");
    promo[i].age    = get_int("Age : ");
    promo[i].moyenne = get_float("Moyenne : ");
    (*\trou[{promo[i].adresse.rue}]{6cm}*)        = get_string("Rue : ");
    (*\trou[{promo[i].adresse.codePostal}]{6cm}*) = get_int("Code postal : ");
    (*\trou[{promo[i].adresse.ville}]{6cm}*)      = get_string("Ville : ");
}
\end{lstlisting}

    \begin{lstlisting}[language=c,escapeinside={(*}{*)},caption={Comptage par ville, à compléter}]
int compter_ville(T_EtudiantAdresse promo[], int nbEtudiants, const char *ville) {
    int i, compteur = 0;

    for (i = 0; i < nbEtudiants; i++) {
        if ((*\trou[{strcmp(promo[i].adresse.ville, ville) == 0}]{10cm}*)) {
            (*\trou[compteur++;]{6cm}*)
        }
    }
    return compteur;
}
\end{lstlisting}

    \UPSTIquestion{Pourquoi la fonction \texttt{compter\_ville} doit-elle recevoir \texttt{nbEtudiants} en paramètre, en plus du tableau \texttt{promo} ?}

    Parce qu'un tableau \trou[ne connaît pas lui-même le nombre d'éléments réellement utilisés (ici TAILLE\_MAX\_PROMO est une taille maximale, pas forcément le nombre d'étudiants effectivement saisis)]{13.5cm} :
    sans cette information, la fonction ne pourrait pas savoir où arrêter sa boucle.
\end{UPSTIexercice}

\begin{UPSTIexercice}{Pour aller plus loin : filtrer et afficher par ville}
    On souhaite maintenant afficher la \textbf{liste complète} des étudiants d'une ville donnée
    (prénom et moyenne), puis calculer la \textbf{moyenne de la classe restreinte aux étudiants
    d'une ville donnée}. On réutilise le tableau \texttt{promo} et la fonction \texttt{compter\_ville}
    de l'exercice précédent.

    \UPSTIquestion{Compléter la fonction \texttt{afficher\_ville} qui affiche le prénom et la moyenne de chaque étudiant d'une ville donnée, et la fonction \texttt{moyenne\_ville} qui renvoie la moyenne de ces étudiants.}

    \begin{lstlisting}[language=c,escapeinside={(*}{*)},caption={Filtrage par ville, à compléter}]
void afficher_ville(T_EtudiantAdresse promo[], int nbEtudiants, const char *ville) {
    int i;

    for (i = 0; i < nbEtudiants; i++) {
        if ((*\trou[{strcmp(promo[i].adresse.ville, ville) == 0}]{10cm}*)) {
            printf("%s : %.2f\n", (*\trou[{promo[i].prenom}]{6cm}*), (*\trou[{promo[i].moyenne}]{6cm}*));
        }
    }
}

float moyenne_ville(T_EtudiantAdresse promo[], int nbEtudiants, const char *ville) {
    int i, nb = 0;
    float somme = 0.0f;

    for (i = 0; i < nbEtudiants; i++) {
        if (strcmp(promo[i].adresse.ville, ville) == 0) {
            (*\trou[{somme += promo[i].moyenne;}]{9cm}*)
            (*\trou[nb++;]{6cm}*)
        }
    }

    if ((*\trou[nb == 0]{6cm}*)) {
        return 0.0f;
    }
    return (*\trou[somme / nb]{6cm}*);
}
\end{lstlisting}

    \UPSTIquestion{Dans le \texttt{main}, écrire l'appel permettant d'afficher les étudiants de la ville saisie par l'utilisateur, puis leur moyenne.}

    \begin{lstlisting}[language=c,escapeinside={(*}{*)},caption={Utilisation dans le main, à compléter}]
int main(void) {
    T_EtudiantAdresse promo[TAILLE_MAX_PROMO];
    int nbEtudiants;
    string ville;

    // ... saisie de nbEtudiants et de promo (cf. exercice precedent) ...

    ville = get_string("Ville recherchee : ");

    printf("Etudiants de %s :\n", ville);
    (*\trou[{afficher\_ville(promo, nbEtudiants, ville);}]{12cm}*)

    printf("Moyenne a %s : %.2f\n", ville,
           (*\trou[{moyenne\_ville(promo, nbEtudiants, ville)}]{10cm}*));
    return 0;
}
\end{lstlisting}
\end{UPSTIexercice}
