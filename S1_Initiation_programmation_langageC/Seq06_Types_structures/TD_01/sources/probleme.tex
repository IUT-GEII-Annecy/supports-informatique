\begin{UPSTIexercice}{Projet d'ensemble : bulletins d'une classe}
    On souhaite écrire un petit programme qui gère les \textbf{bulletins} d'une classe.
    Chaque élève a plusieurs notes dans différentes matières, avec un \textbf{coefficient}
    pour chaque matière.

    On suppose que :
    \begin{itemize}
        \item tous les élèves ont les \textbf{mêmes matières}, dans le même ordre,
        \item il y a au maximum \texttt{30} élèves dans la classe,
        \item il y a par exemple \texttt{4} matières par élève,
        \item chaque élève peut avoir \textbf{plusieurs notes} par matière (par ex. contrôles, devoirs, etc.).
    \end{itemize}

    \UPSTIquestion{Compléter les constantes et les types structurés permettant de modéliser ce problème : une structure \texttt{T\_Matiere} (nom, coefficient) et une structure \texttt{T\_Eleve} (prénom, âge, notes par matière, et le nombre de notes réellement saisies par matière).}

    \begin{lstlisting}[language=c,escapeinside={(*}{*)},caption={Modélisation, à compléter}]
#define TAILLE_MAX_CLASSE 30
#define NB_MATIERES       4
#define NB_NOTES_MAX      10

typedef struct {
    (*\trou[string nom;]{7.5cm}*)
    (*\trou[float coeff;]{7.5cm}*)
} T_Matiere;

typedef struct {
    (*\trou[{float notes[NB\_MATIERES][NB\_NOTES\_MAX];}]{11.5cm}*)
    (*\trou[{int nb\_notes[NB\_MATIERES];}]{9cm}*)  // nombre de notes pour chaque matiere
    (*\trou[string prenom;]{7.5cm}*)
    (*\trou[int age;]{7.5cm}*)
} T_Eleve;

T_Matiere matieres[NB_MATIERES];
\end{lstlisting}

    Cette modélisation sépare les définitions des matières (nom et coefficient, partagés par tous)
    du stockage des notes individuelles de chaque élève. Chaque élève peut avoir un nombre
    différent de notes par matière, stocké dans \texttt{nb\_notes[]}.

    % ----------------------------------------------------------------------

    \UPSTIquestion{Compléter le corps de la fonction \texttt{float moyenne\_eleve(T\_Eleve e, T\_Matiere matieres[])} qui renvoie la moyenne générale de l'élève, en tenant compte des coefficients de chaque matière.}

    \begin{lstlisting}[language=c,escapeinside={(*}{*)},caption={moyenne\_eleve, à compléter}]
float moyenne_eleve(T_Eleve e, T_Matiere matieres[]) {
    int i, k;
    float somme_notes = 0.0f;
    float somme_coeffs = 0.0f;

    for (i = 0; i < NB_MATIERES; i++) {
        float moy_matiere = 0.0f;

        // Calculer la moyenne pour cette matiere
        for (k = 0; k < e.nb_notes[i]; k++) {
            (*\trou[{moy\_matiere += e.notes[i][k];}]{9.5cm}*)
        }
        if (e.nb_notes[i] > 0) {
            moy_matiere /= e.nb_notes[i];
        }

        // Ajouter a la moyenne ponderee
        (*\trou[{somme\_notes  += moy\_matiere * matieres[i].coeff;}]{9.5cm}*)
        (*\trou[{somme\_coeffs += matieres[i].coeff;}]{9.5cm}*)
    }

    if (somme_coeffs == 0.0f) {
        return 0.0f;
    }
    return (*\trou[{somme\_notes / somme\_coeffs;}]{9.5cm}*)
}
\end{lstlisting}

    % ----------------------------------------------------------------------
    On se propose d'écrire la fonction \texttt{float moyenne\_matiere(T\_Eleve classe[], int nbEleves, int indiceMatiere)} qui renvoie la moyenne de la classe pour une matière donnée, identifiée par son indice.

    \UPSTIquestion{\texttt{moyenne\_eleve} n'a pas besoin du nombre de matières en paramètre, alors
        que \texttt{moyenne\_matiere} a besoin de \texttt{nbEleves}. Pourquoi cette différence ?}

    Le nombre de matières par élève est une \trouTexte[constante (NB\_MATIERES), connue à la compilation]{6cm},
    alors que le nombre d'élèves réellement saisis est une \trouTexte[valeur saisie à l'exécution, que le
    tableau classe ne connaît pas lui-même (TAILLE\_MAX\_CLASSE n'est qu'une taille maximale)]{9cm} :
    sans ce paramètre, la fonction ne saurait pas où arrêter sa boucle.

    \UPSTIquestion{Compléter le corps de \texttt{moyenne\_matiere}, sur le même principe que
        \texttt{moyenne\_eleve} (accumuler une somme et un nombre de notes, puis diviser).}

    \begin{lstlisting}[language=c,escapeinside={(*}{*)},caption={moyenne\_matiere, à compléter}]
float moyenne_matiere(T_Eleve classe[], int nbEleves, int indiceMatiere) {
    int i, k;
    float somme = 0.0f;
    int nb_notes_total = 0;

    for (i = 0; i < (*\trou[nbEleves]{6.5cm}*); i++) {
        for (k = 0; k < classe[i].nb_notes[indiceMatiere]; k++) {
            (*\trou[{somme += classe[i].notes[indiceMatiere][k];}]{9.5cm}*)
            (*\trou[nb\_notes\_total++;]{7.5cm}*)
        }
    }

    if (nb_notes_total == 0) {
        return 0.0f;
    }
    return somme / nb_notes_total;
}
\end{lstlisting}

    % ----------------------------------------------------------------------
    On écrit maintenant le \texttt{main}, étape par étape. On dispose d'un tableau
    \texttt{T\_Eleve classe[TAILLE\_MAX\_CLASSE];} et d'un tableau \texttt{T\_Matiere matieres[NB\_MATIERES];}.

    \UPSTIquestion{Compléter le code qui demande le nombre d'élèves réel de la classe (entre \texttt{1} et
        \texttt{TAILLE\_MAX\_CLASSE}), puis saisit, pour chaque matière du tableau \texttt{matieres}, son nom
        et son coefficient.}

\begin{lstlisting}[language=c,escapeinside={(*}{*)},caption={Saisie du nombre d'élèves et des matières, à compléter}]
int nbEleves;
int j;

// Saisie du nombre d'eleves
do {
    nbEleves = get_int("Nombre d'eleves (1..%d) : ", TAILLE_MAX_CLASSE);
} while ((*\trou[{nbEleves < 1 || nbEleves > TAILLE\_MAX\_CLASSE}]{11.5cm}*));

// Saisie des matieres
for (j = 0; j < NB_MATIERES; j++) {
    printf("Matiere %d\n", j + 1);
    (*\trou[{matieres[j].nom}]{7.5cm}*)   = get_string("Nom de la matiere : ");
    (*\trou[{matieres[j].coeff}]{7.5cm}*) = get_float("Coefficient : ");
}
\end{lstlisting}

    % ----------------------------------------------------------------------

    \UPSTIquestion{Compléter le code qui, pour chaque élève de la classe (indice \texttt{i} de \texttt{0} à
        \texttt{nbEleves - 1}), saisit son prénom, son âge, puis pour chaque matière le nombre de notes
        de l'élève dans cette matière et chacune de ces notes.}

\begin{lstlisting}[language=c,escapeinside={(*}{*)},caption={Saisie d'un élève et de ses notes, à compléter}]
int i, j, k;

for (i = 0; i < nbEleves; i++) {
    printf("Eleve %d\n", i + 1);
    (*\trou[{classe[i].prenom}]{7.5cm}*) = get_string("Prenom : ");
    (*\trou[{classe[i].age}]{7.5cm}*)    = get_int("Age : ");

    for (j = 0; j < NB_MATIERES; j++) {
        (*\trou[{classe[i].nb\_notes[j]}]{9.5cm}*) = get_int(
            "Nombre de notes en %s : ", matieres[j].nom);

        for (k = 0; k < (*\trou[{classe[i].nb\_notes[j]}]{9.5cm}*); k++) {
            (*\trou[{classe[i].notes[j][k]}]{9.5cm}*) = get_float(
                "Note %d : ", k + 1);
        }
    }
    printf("\n");
}
\end{lstlisting}

    % ----------------------------------------------------------------------

    \UPSTIquestion{En utilisant \texttt{moyenne\_eleve}, compléter le code qui affiche la moyenne générale de
        chaque élève de la classe.}

\begin{lstlisting}[language=c,escapeinside={(*}{*)},caption={Affichage des moyennes, à compléter}]
int i;

for (i = 0; i < nbEleves; i++) {
    float m = (*\trou[{moyenne\_eleve(classe[i], matieres)}]{11.5cm}*);
    printf("Moyenne de %s = %.2f\n", classe[i].prenom, m);
}
\end{lstlisting}

    % ----------------------------------------------------------------------

    \UPSTIquestion{En utilisant \texttt{moyenne\_eleve}, et sur le même principe que la recherche du
        meilleur étudiant du premier exercice, compléter le code qui détermine l'élève ayant la
        meilleure moyenne générale de la classe, et affiche son prénom et sa moyenne.}

\begin{lstlisting}[language=c,escapeinside={(*}{*)},caption={Meilleur élève, à compléter}]
int indice_meilleur = 0;
float meilleure_moy = moyenne_eleve(classe[0], matieres);
float m;
int i;

for (i = 1; i < nbEleves; i++) {
    m = moyenne_eleve(classe[i], matieres);
    if ((*\trou[{m > meilleure\_moy}]{9.5cm}*)) {
        meilleure_moy = m;
        indice_meilleur = i;
    }
}

printf("Meilleur eleve : %s (moyenne = %.2f)\n",
       (*\trou[{classe[indice\_meilleur].prenom}]{7.5cm}*), meilleure_moy);
\end{lstlisting}

    % ----------------------------------------------------------------------

    \UPSTIquestion{En utilisant \texttt{moyenne\_matiere}, compléter le code qui calcule et affiche la moyenne
        de la classe pour chaque matière, puis détermine et affiche la matière dont la moyenne de classe
        est la plus faible (matière « à renforcer »). La recherche du minimum se fait exactement comme la
        recherche du maximum à la question précédente, en inversant le sens de la comparaison.}

\begin{lstlisting}[language=c,escapeinside={(*}{*)},caption={Matière à renforcer, à compléter}]
float moyennesMatieres[NB_MATIERES];
int j;

for (j = 0; j < NB_MATIERES; j++) {
    moyennesMatieres[j] = (*\trou[{moyenne\_matiere(classe, nbEleves, j)}]{13.5cm}*);
    printf("Moyenne de la classe en %s : %.2f\n",
           matieres[j].nom, moyennesMatieres[j]);
}

// Recherche de la matiere avec la plus faible moyenne
int indice_pire_matiere = 0;
float pire_moyenne = moyennesMatieres[0];

for (j = 1; j < NB_MATIERES; j++) {
    if ((*\trou[{moyennesMatieres[j] < pire\_moyenne}]{11.5cm}*)) {
        pire_moyenne = moyennesMatieres[j];
        indice_pire_matiere = j;
    }
}

printf("Matiere a renforcer : %s (moyenne = %.2f)\n",
       (*\trou[{matieres[indice\_pire\_matiere].nom}]{9.5cm}*), pire_moyenne);
\end{lstlisting}

    Cet ensemble de questions mobilise plusieurs \textbf{structures} (données partagées et individuelles),
    des \textbf{tableaux de structures}, des \textbf{fonctions} qui prennent des structures et des tableaux
    de structures en paramètre, ainsi que des \textbf{boucles imbriquées} et un peu d'algorithmique
    (recherche de max/min).
\end{UPSTIexercice}

\begin{UPSTIexercice}{Conception libre : à vous de choisir la structure}
    Jusqu'ici, les champs des structures étaient toujours donnés. On va maintenant
    \textbf{concevoir soi-même} une structure adaptée à un problème, sans qu'on nous dise quels
    champs utiliser.

    \UPSTIquestion{Lire le cahier des charges suivant.}

    On veut représenter, dans une chaîne d'acquisition, l'\textbf{historique des relevés} d'un
    capteur. Cahier des charges :
    \begin{itemize}
        \item un capteur est caractérisé par une référence et l'unité physique de la grandeur
            qu'il mesure (par ex. \texttt{"degC"}, \texttt{"V"}, \texttt{"bar"}) ;
        \item un capteur effectue plusieurs \textbf{relevés} au cours du temps : on veut pouvoir
            stocker un \textbf{historique} de relevés (jusqu'à une taille maximale fixée), pas
            une seule mesure ;
        \item chaque relevé associe une valeur mesurée à l'instant de la mesure (heure, minute,
            seconde).
    \end{itemize}

    Certaines de ces informations vont naturellement ensemble : à vous d'identifier lesquelles, et
    de décider si une \textbf{structure imbriquée} est pertinente (comme on l'a fait pour
    \texttt{T\_Adresse} dans \texttt{T\_EtudiantAdresse}).

    \UPSTIquestion{Proposer et écrire en C le ou les types structurés (\texttt{typedef struct})
        nécessaires pour respecter le cahier des charges ci-dessus. Vous êtes libre des noms de
        champs et de leurs types, du nombre de structures (une seule, ou plusieurs imbriquées), et
        de la constante de taille maximale de l'historique. Puis, en utilisant ce type, écrire le
        corps de la fonction suivante, dont le prototype est donné :
        \begin{lstlisting}[language=c,caption={Prototype de la fonction demandée}]
// Ajoute le releve r a l'historique du capteur c, si la place le permet,
// et renvoie le capteur mis a jour.
T_Capteur ajouter_releve(T_Capteur c, T_Releve r);
        \end{lstlisting}}

\end{UPSTIeleveOnlyEnv}

\begin{UPSTIcorrectionP}{Exemple de solution}
    Il n'y a pas de solution unique : voici un exemple cohérent, avec une structure imbriquée pour
    l'horodatage d'un relevé, un tableau de structures pour l'historique des relevés du capteur, et
    la fonction \texttt{ajouter\_releve} demandée.
\begin{lstlisting}[language=c,caption={Exemple de correction}]
#define TAILLE_MAX_HISTORIQUE 100

typedef struct {
    int h;
    int m;
    int s;
} T_Horodatage;

typedef struct {
    float valeur;
    T_Horodatage instant;
} T_Releve;

typedef struct {
    string reference;
    string unite;                           // par ex. "degC", "V", "bar"
    T_Releve historique[TAILLE_MAX_HISTORIQUE];
    int nbReleves;      // nombre de releves reellement presents dans l'historique
} T_Capteur;

// La structure n'est pas modifiable par une fonction "classique" sans
// pointeur : on renvoie le capteur modifie, que l'appelant doit
// reaffecter (c = ajouter_releve(c, r);).
T_Capteur ajouter_releve(T_Capteur c, T_Releve r) {
    if (c.nbReleves < TAILLE_MAX_HISTORIQUE) {
        c.historique[c.nbReleves] = r;
        c.nbReleves++;
    }
    return c;
}
\end{lstlisting}
\end{UPSTIcorrectionP}
\end{UPSTIexercice}
