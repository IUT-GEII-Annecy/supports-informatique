
\section{Nos premières boucles While, For et Do...While}

%\subsection{La boucle While - Miauler n fois}

\begin{UPSTIexercice}{Boucle while : Un chat qui miaule}
	Le programme \texttt{miauler.c} contient le code suivant, qui fait miauler un chat 3 fois :
	\begin{lstlisting}[language=c]
#include <stdio.h>
int main(void) {
    int i = 0;
    while (i < 3) {
        printf("Miaou !\n");
        i++;
    }
    return 0;
}
\end{lstlisting}
	\UPSTIetape Tester le programme miauler.c
	\UPSTIetape Modifier le programme pour qu'il miaule 5 fois.
	\UPSTIetape Modifier le programme pour qu'il miaule \texttt{n} fois, où \texttt{n} est un entier entré par l'utilisateur.
\end{UPSTIexercice}


%\subsection{La boucle For - Aboyer n fois}

\begin{UPSTIexercice}{Boucle for : Un chien qui aboie}
	Le programme \texttt{aboyer.c} contient le code suivant, qui fait aboyer un chien 3 fois :
	\begin{lstlisting}[language=c]
#include <stdio.h>
int main(void) {
    for (int i = 0; i < 3; i++) {
        printf("Ouaf !\n");
    }
    return 0;
}
\end{lstlisting}
	\UPSTIetape Tester le programme aboyer.c
	\UPSTIetape Modifier le programme pour qu'il aboie 5 fois.
	\UPSTIetape Modifier le programme pour qu'il aboie \texttt{n} fois, où \texttt{n} est un entier entré par l'utilisateur.
\end{UPSTIexercice}


%\subsection{La boucle Do...While - Chanter n fois}

\begin{UPSTIexercice}{Boucle do...while Un oiseau qui chante}
	Le programme \texttt{chanter.c} contient le code suivant, qui fait chanter un oiseau 3 fois :
	\begin{lstlisting}[language=c]
#include <stdio.h>
int main(void) {
    int i = 0;
    do {
        printf("Cui-cui !\n");
        i++;
    } while (i < 3);
    return 0;
}
\end{lstlisting}
	\UPSTIetape Tester le programme chanter.c
	\UPSTIetape Modifier le programme pour qu'il chante 5 fois.
	\UPSTIetape Modifier le programme pour qu'il chante \texttt{n} fois, où \texttt{n} est un entier entré par l'utilisateur.
\end{UPSTIexercice}
