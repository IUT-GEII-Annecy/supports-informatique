% ---- boucle-ecrire ----
\begin{question}{ent-boucle-ecrire}
    Écrire une boucle en langage C qui affiche les multiples de 5 de 5 à 25 inclus (une valeur par ligne).
    \evaluationProfGrille{2}{9}
\end{question}

\begin{UPSTIcorrectionP}{boucle-ecrire}
    \begin{lstlisting}[language=c]
for (int i = 5; i <= 25; i = i + 5) {
    printf("%d\n", i);
}
    \end{lstlisting}
\end{UPSTIcorrectionP}
